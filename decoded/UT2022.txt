       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Srv_Name   = TrimAndCheck(%trimr(Srv_Name  ) : *IN95);
           Srv_Desc   = TrimAndCheck(%trimr(Srv_Desc  ) : *IN95);
           @SrvMatch  = TrimAndCheck(%trimr(@SrvMatch ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Srv_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must meet configuration standards

           If @ErrorMsg = *blank and
               ConfigValidName(Srv_Name) <> *ON;
               @ErrorMsg = 'Name does not meet configuration standards';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigServerID(Srv_Name) <> *zero and
               ConfigServerID(Srv_Name) <> @Id) or
               (@Option <> 2 and
               ConfigServerID(Srv_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Srv_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       // Match Value is required

           If @ErrorMsg = *blank and
               @SrvMatch = *blank;
               @ErrorMsg = 'Match Value is a required entry';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Load screen values to database values

           Srv_Info = %trim(@Info);
           Srv_Match  = %Trim(@SrvMatch);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigServerInsert(@DB);
           ElseIf @Option = 2;
               ConfigServerUpdate(@DB);
           ElseIf @Option = 4;
               ConfigServerDelete(@DB);
           Endif;

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Prompt;

       // check for prompt on field where cursor is positioned

          @ErrorMsg = 'Prompt function not supported or defined ' +
                      'at current cursor position';
    
       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Check_Function_Key_Requests;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
               @DB = ConfigServerGet(@Id);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGSRV';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';

       // Check for delete with child records

           If @Option = 4 and
              ConfigServerKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info      = Srv_Info;
           @SrvMatch  = Srv_Match;

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
