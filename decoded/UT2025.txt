       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Xtn_Name   = TrimAndCheck(%trimr(Xtn_Name  ) : *IN95);
           Xtn_Desc   = TrimAndCheck(%trimr(Xtn_Desc  ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Xtn_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must meet configuration standards

           If @ErrorMsg = *blank and
               ConfigValidName(Xtn_Name) <> *ON;
               @ErrorMsg = 'Name does not meet configuration standards';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigExtentID(@ParentID : Xtn_Name) <> *zero and
               ConfigExtentID(@ParentID : Xtn_Name) <> @Id) or
               (@Option <> 2 and
               ConfigExtentID(@ParentID : Xtn_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Xtn_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Prime parent key value

           Ctx_ID = @ParentID;

       // Load screen values to database values

           Xtn_Info = %trim(@Info);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigExtentInsert(@DB);
           ElseIf @Option = 2;
               ConfigExtentUpdate(@DB);
           ElseIf @Option = 4;
               ConfigExtentDelete(@DB);
           Endif;

       Endsr;
    

                       'at current cursor position';

       Endsr;

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
               @DB = ConfigExtentGet(@Id);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGXTN';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';

       // Check for delete with child records

           If @Option = 4 and
              ConfigExtentKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info      = Xtn_Info;

       // Set header values

           @Ctx_Title = ConfigContextTitle(@ParentID);

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
