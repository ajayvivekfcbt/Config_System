delete from u4142ap.BLROLLUPD;

INSERT INTO u4142ap.BLROLLUPD
WITH t1 AS
(SELECT 'BLGLBLHS',GLH_TSP_REC_UPDATE FROM BLGLBLHS ORDER BY
 GLH_TSP_REC_UPDATE DESC FETCH FIRST  ROW ONLY )
 ,   t2 AS
(SELECT 'BLGLENTR',GLE_TSP_REC_UPDATE FROM BLGLENTR ORDER BY
 GLE_TSP_REC_UPDATE DESC FETCH FIRST  ROW ONLY )
, T3 AS
(SELECT 'BLOSTTRN',OTR_TSP_REC_UPDATE FROM BLOSTTRN ORDER BY
OTR_TSP_REC_UPDATE DESC FETCH FIRST  ROW ONLY)
,T4 AS
(SELECT 'BLTRNEVT',TRE_TSP_REC_UPDATE FROM BLTRNEVT ORDER BY
TRE_TSP_REC_UPDATE DESC FETCH FIRST  ROW ONLY)
,T5 AS
(SELECT 'BLOBJEXT',OEA_UPDTSP FROM BLOBJEXT ORDER BY
OEA_UPDTSP DESC FETCH FIRST ROW ONLY)
SELECT * FROM t1
UNION
SELECT * FROM t2
UNION
SELECT * FROM t3
UNION
SELECT * FROM t4
UNION
SELECT * FROM t5 ;

MERGE INTO BLROLL
USING U4142AP.BLROLLUPD
ON TGTFILE = HVR00001
WHEN MATCHED THEN UPDATE SET LASTRUNDT = GLH_UPDTSP;
