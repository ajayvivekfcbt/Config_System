      *--------------------------------------------------------------------
      * Define specific prototypes
      *--------------------------------------------------------------------
     D WorkWithVariables...
     D                 PR                  EXTPGM('UT2063')
     D                                9B 0 const                                Scope ID
     D                                1A                                        End FKey

     D  @Scp_ID        S              9B 0                                      Scope ID
     D  @EndFKey       S              1A                                        End FKey

      /Copy QCPYSRC,UTCFG_PR
      /Copy QCPYSRC,UTBIND_PR

     *********************************************************************
     **        M A I N L I N E    C O N T R O L    S E C T I O N
     **********************************************************************
      /FREE

       // Resolve to current environment

       @Scp_ID = ConfigScopeID(ConfigServerID(ConfigServer) :
                               ConfigEnvScope());

       // Send message if no scope found

       If @Scp_ID = *zero;
          SendEscapeMessage('Unable to resolve to a defined Server/Scope');
       Endif;

       // Call work with variables program

       WorkWithVariables(@Scp_ID : @EndFKey);

       // End of program

       *INLR = *On;
