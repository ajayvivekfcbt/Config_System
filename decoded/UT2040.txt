     D EntryParms      PI
     D  @EntryID                      9B 0                                      Input ID
     D  @EndFKey                      1A                                        Return Function Key

     D MaintainOneRow...
     D                 PR                  EXTPGM('UT2026')
     D                                9B 0 const                                Parent ID

     D                                1A                                        End FKey

       Begsr $Load_Subfile;

       // Retrieve the data for display

           EXEC SQL
               DECLARE LIST_CURSOR CURSOR FOR
               SELECT   VDF.VDF_ID, VDF.XTN_ID,
                        TRIM(CTX.CTX_NAME) || '.' ||
                           TRIM(XTN.XTN_NAME) || '.'||
                           TRIM(VDF.VDF_NAME),
                           VDF.VDF_DESC
               FROM     UTCFGVDF VDF
                            JOIN
                        UTCFGXTN XTN ON XTN.XTN_ID = VDF.XTN_ID
                            JOIN
                        UTCFGCTX CTX ON CTX.CTX_ID = XTN.CTX_ID
               WHERE    VDF.VTP_ID = :@EntryID
               ORDER BY 3;

           EXEC SQL OPEN LIST_CURSOR;

           DOU SQLCOD <> *zero;

               EXEC SQL
                   FETCH LIST_CURSOR
                   INTO  :@ID, :@PARENTID, :@VARIABLE, :@DESC;

               If SQLCOD = *zero;
                   Exsr $Write_Subfile_Record;
               Endif;
           Enddo;

           EXEC SQL CLOSE LIST_CURSOR;

       Endsr;
