     H OPTION(*SRCSTMT : *NODEBUGIO) CVTOPT(*NODATETIME) ALWNULL(*USRCTL)
     H DFTACTGRP(*NO) ACTGRP(*CALLER) BNDDIR('QC2LE' : 'UTBIND')
      // -------------------------------------------------------------------------------------------
      //
      // Program  : UT2080 - Stage Remote Variable Type Definitions
      // Project  : Infrastructure - Configuration
      // Author.  : G.Neale                          Date. . : 11/26/2011
      //
      // Narrative: Create and load UTCFGVTP from remote system
      //
      // -------------------------------------------------------------------------------------------
      // Modification History:
      //--------------------------------------------------------------------------------------------
      //
      // -------------------------------------------------------------------------------------------

      /Copy QCPYSRC,PGSDS

      *--------------------------------------------------------------------
      * Define entry prototype and procedure interface
      *--------------------------------------------------------------------
     D EntryParms      PR                  EXTPGM('UT2080')
     D                               20A                                        Remote Server

     D EntryParms      PI
     D   @Remote                     20A                                        Remote Server

      /Copy QCPYSRC,UTBIND_PR

      *--------------------------------------------------------------------
      * Define variables
      *--------------------------------------------------------------------
     D  @MsgID         S              7A
     D  @Local         S             20A

     D  Vtp_ID         S              9B 0
     D  Vtp_Name       S             20A
     D  Vtp_Desc       S             50A
     D  Vtp_Proc       S            128A
     D  Vtp_Info       S           2048A

     **********************************************************************
     **        M A I N L I N E    C O N T R O L    S E C T I O N
     **********************************************************************
      /FREE

       // Set current server name

        Exec Sql
           Select   Current Server
           Into     :@Local
           From     Sysibm/SysDummy1;

       // Drop local temp table

       EXEC SQL SET CONNECTION :@Local;

       Exec SQL Drop Table QTEMP/TEMP_VTP;

       // Create local temp table

       Exec SQL Create Table QTEMP/TEMP_VTP Like UTCFGVTP;

       // Create SQL package prior to execution

       RunCommand('CRTSQLPKG PGM(' + %trim(#PS.ProgramName) +
                    ') RDB(' + %trim(@Remote) + ')');

       // Define and open cursor

       EXEC SQL DECLARE RemoteTable CURSOR FOR
          SELECT   VTP_ID,
                   VTP_NAME,
                   VTP_DESC,
                   TRIM(VTP_INFO),
                   TRIM(VTP_INFO)
           FROM    UTCFGVTP;

       EXEC SQL OPEN RemoteTable;

       Dou SQLCod <> *zero;
          EXEC SQL FETCH RemoteTable
                   INTO :Vtp_ID, :Vtp_Name, :Vtp_Desc, :Vtp_Proc, :Vtp_Info;
          If SQLCod = *zero;
             EXEC SQL SET CONNECTION :@Local;
             Exec SQL
                Insert   into TEMP_VTP
                Values   (
                         :Vtp_ID,
                         :Vtp_Name,
                         :Vtp_Desc,
                         TRIM(:Vtp_Proc),
                         TRIM(:Vtp_Info));
              EXEC SQL SET CONNECTION :@Remote;
           Endif;
       Enddo;

       EXEC SQL CLOSE RemoteTable;

       // Disconnect from FCB and repoint to DEV

       EXEC SQL DISCONNECT :@Remote;
       EXEC SQL SET CONNECTION :@Local;
       If  SQLCOD = -843;
           EXEC SQL CONNECT RESET;
       Endif;

       // End of program

       *INLR = *On;

      /end-free
