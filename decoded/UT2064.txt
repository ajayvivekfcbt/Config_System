     D EntryParms      PR                  EXTPGM('UT2064')
     D                                9B 0 const                                Server ID

     D                                1A                                        End FKey

     D EntryParms      PI
     D   @ServerID                    9B 0 const                                Entry ID
     D   @ScopeID1                    9B 0 const                                Entry ID
     D   @EndFKey                     1A                                        End FKey

     D CompareVariables...
     D                 PR                  EXTPGM('UT2065')
     D                                9B 0 const                                First Scope

       Begsr $Load_Subfile;

       // Retrieve the data for display

           EXEC SQL
               DECLARE LIST_CURSOR CURSOR FOR
               SELECT   SCP.SCP_ID, SCP.SCP_NAME, SCP.SCP_DESC,
                        COALESCE(SRM.SRM_NAME,' ')
               FROM     UTCFGSCP SCP
                            JOIN
                        UTCFGSRM SRM ON SRM.SRM_ID = SCP.SRM_ID
               WHERE    SCP.SRV_ID = :@ServerID AND
                        SCP.SCP_ID <> :@ScopeID1 AND
                        SCP.SCP_IS_SRV = 0
               ORDER BY UPPER(SCP.SCP_NAME);

           EXEC SQL OPEN LIST_CURSOR;

           DOU SQLCOD <> *zero;

               EXEC SQL
                   FETCH LIST_CURSOR
                   INTO  :@ID, :@NAME, :@DESC, :@SRMNAME;

               If SQLCOD = *zero;
                   Exsr $Write_Subfile_Record;
               Endif;
           Enddo;

           EXEC SQL CLOSE LIST_CURSOR;

       Endsr;
