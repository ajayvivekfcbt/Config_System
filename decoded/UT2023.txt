      *--------------------------------------------------------------------
     D EntryParms      PR                  EXTPGM('UT2023')
     D                                9B 0 const                                Parent ID
     D                                2S 0 const                                Option Number
     D                                9B 0 const                                Row ID
     D                                1A                                        End FKey

     D EntryParms      PI
     D   @ParentID                    9B 0 const                                Parent ID
     D   @Option                      2S 0 const                                Option Number
     D   @ID                          9B 0 const                                Row ID
     D   @EndFKey                     1A                                        End FKey

      /Copy QCPYSRC,UTCFG_PR

      *--------------------------------------------------------------------
      * Define database work structure
      *--------------------------------------------------------------------
     D @DB           E DS                  ExtName(UTCFGSCP) INZ

      *--------------------------------------------------------------------
      * Define prototypes
      *--------------------------------------------------------------------
     D PromptMethod...
     D                 PR                  EXTPGM('UT2031')
     D                                9B 0                                      Row ID

      *--------------------------------------------------------------------
      * Define specific variables
      *--------------------------------------------------------------------
     D @ReturnID       S              9B 0

      /Copy QCPYSRC,UTCOM_RM
      /FREE

       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Scp_Name   = TrimAndCheck(%trimr(Scp_Name  ) : *IN95);
           Scp_Desc   = TrimAndCheck(%trimr(Scp_Desc  ) : *IN95);
           @ScpMatch  = TrimAndCheck(%trimr(@ScpMatch ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Scp_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must meet configuration standards

           If @ErrorMsg = *blank and
               ConfigValidName(Scp_Name) <> *ON;
               @ErrorMsg = 'Name does not meet configuration standards';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigScopeID(@ParentID : Scp_Name) <> *zero and
               ConfigScopeID(@ParentID : Scp_Name) <> @Id) or
               (@Option <> 2 and
               ConfigScopeID(@ParentID : Scp_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Scp_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       // Server level must be YES or NO

           If @ErrorMsg = *blank and
               @YN_Is_Srv <> 'YES' and
               @YN_Is_Srv <> 'NO';
               @ErrorMsg = 'Server Level must be YES or NO';
           Endif;

       // Scope Resolution Method must be valid

           If @ErrorMsg = *blank;
               Srm_ID = ConfigMethodID(Srm_Name);
               If Srm_ID = *zero;
                   @ErrorMsg = 'Scope Resolution Method is invalid';
               Else;
                   Srm_Name = ConfigMethodName(Srm_ID);
                   Srm_Desc = ConfigMethodDesc(Srm_ID);
               Endif;
           Endif;

       // Match Value is required

           If @ErrorMsg = *blank and
               @ScpMatch = *blank;
               @ErrorMsg = 'Match Value is a required entry';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Load screen values to database values

           Srv_ID = @ParentID;
           Scp_Info = %trim(@Info);
           Scp_Is_Srv = YesNoToBit(@YN_Is_Srv);
           Scp_Match  = %Trim(@ScpMatch);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigScopeInsert(@DB);
           ElseIf @Option = 2;
               ConfigScopeUpdate(@DB);
           ElseIf @Option = 4;
               ConfigScopeDelete(@DB);
           Endif;

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Prompt;

       // check for prompt on field where cursor is positioned

           If @PromptFld = 'SRM_NAME';
               PromptMethod(@ReturnID);
               If  @ReturnID <> *zero;
                   Srm_ID = @ReturnID;
                   Srm_Name = ConfigMethodName(Srm_ID);
                   Srm_Desc = ConfigMethodDesc(Srm_ID);
               Endif;
           Else;
               @ErrorMsg = 'Prompt function not supported or defined ' +
                           'at current cursor position';
           Endif;

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Check_Function_Key_Requests;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $Data_Initialization;

       // If @Id received - get row and populate foreign values

           If @Id <> *Zero;
              @DB = ConfigScopeGet(@Id);
              Srm_Name = ConfigMethodName(Srm_ID);
              Srm_Desc = ConfigMethodDesc(Srm_ID);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGSCP';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';
           If @Option = *zero or
              @Option = 2 or
              @Option = 3;
              @AvlFnc(4) = 'F4=Prompt';
           Endif;

       // Check for delete with child records

           If @Option = 4 and
              ConfigScopeKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info      = Scp_Info;
           @ScpMatch  = Scp_Match;
           @YN_Is_Srv = BitToYesNo(Scp_Is_Srv);

       // Set header values

           @Srv_Title = ConfigServerTitle(@ParentID);

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
