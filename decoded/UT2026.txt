     H OPTION(*SRCSTMT : *NODEBUGIO) CVTOPT(*NODATETIME) ALWNULL(*USRCTL)
     H DFTACTGRP(*NO) ACTGRP(*CALLER)
     H BNDDIR('QC2LE' : 'UTBIND' : 'UTCFG')

     FUT2026D   CF   E             WORKSTN INFDS(#WS)

      *--------------------------------------------------------------------
      * Define entry prototype and procedure interface
      *--------------------------------------------------------------------
     D EntryParms      PR                  EXTPGM('UT2026')
     D                                9B 0 const                                Parent ID
     D                                2S 0 const                                Option Number
     D                                9B 0 const                                Row ID
     D                                1A                                        End FKey

     D EntryParms      PI
     D   @ParentID                    9B 0 const                                Parent ID
     D   @Option                      2S 0 const                                Option Number
     D   @ID                          9B 0 const                                Row ID
     D   @EndFKey                     1A                                        End FKey

      /Copy QCPYSRC,UTCFG_PR

      *--------------------------------------------------------------------
      * Define database work structure
      *--------------------------------------------------------------------
     D @DB           E DS                  ExtName(UTCFGVDF) INZ

      *--------------------------------------------------------------------
      * Define prototypes
      *--------------------------------------------------------------------
     D PromptVarType...
     D                 PR                  EXTPGM('UT2030')
     D                                9B 0                                      Row ID

      *--------------------------------------------------------------------
      * Define specific variables
      *--------------------------------------------------------------------
     D @ReturnID       S              9B 0

      /Copy QCPYSRC,UTCOM_RM
      /FREE

       //--------------------------------------------------------------------

       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Vdf_Name   = TrimAndCheck(%trimr(Vdf_Name  ) : *IN95);
           Vdf_Desc   = TrimAndCheck(%trimr(Vdf_Desc  ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Vdf_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must meet configuration standards

           If @ErrorMsg = *blank and
               ConfigValidName(Vdf_Name) <> *ON;
               @ErrorMsg = 'Name does not meet configuration standards';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigVarDefID(@ParentID : Vdf_Name) <> *zero and
               ConfigVarDefID(@ParentID : Vdf_Name) <> @Id) or
               (@Option <> 2 and
               ConfigVarDefID(@ParentID : Vdf_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Vdf_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       // Restricted values must be YES or NO

           If @ErrorMsg = *blank and
               @Restrict <> 'YES' and
               @Restrict <> 'NO';
               @ErrorMsg = 'Restricted values must be YES or NO';
           Endif;

       // Variable Type must be valid

           If @ErrorMsg = *blank;
               Vtp_ID = ConfigVarTypeID(Vtp_Name);
               If Vtp_ID = *zero;
                   @ErrorMsg = 'Variable Type is invalid';
               Else;
                   Vtp_Name = ConfigVarTypeName(Vtp_ID);
                   Vtp_Desc = ConfigVarTypeDesc(Vtp_ID);
               Endif;
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Prime parent key value

           Xtn_ID = @ParentID;

       // Load screen values to database values

           Vdf_Info = %trim(@Info);
           Vdf_Rstrct = YesNoToBit(@Restrict);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigVarDefInsert(@DB);
           ElseIf @Option = 2;
               ConfigVarDefUpdate(@DB);
           ElseIf @Option = 4;
               ConfigVarDefDelete(@DB);
           Endif;

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Prompt;

       // check for prompt on field where cursor is positioned

           If @PromptFld = 'VTP_NAME';
               PromptVarType(@ReturnID);
               If  @ReturnID <> *zero;
                   Vtp_ID = @ReturnID;
                   Vtp_Name = ConfigVarTypeName(Vtp_ID);
                   Vtp_Desc = ConfigVarTypeDesc(Vtp_ID);
               Endif;
           Else;
               @ErrorMsg = 'Prompt function not supported or defined ' +
                           'at current cursor position';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------
    
       Begsr $Check_Function_Key_Requests;

       Endsr;

       //---------------------------------------------------------------------

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
              @DB = ConfigVarDefGet(@Id);
              Vtp_Name = ConfigVarTypeName(Vtp_ID);
              Vtp_Desc = ConfigVarTypeDesc(Vtp_ID);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGVDF';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';
           If @Option = *zero or
              @Option = 2 or
              @Option = 3;
              @AvlFnc(4) = 'F4=Prompt';
           Endif;

       // Check for delete with child records

           If @Option = 4 and
              ConfigVarDefKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info      = Vdf_Info;
           @Restrict  = BitToYesNo(Vdf_Rstrct);

       // Set header values

           @Xtn_Title = ConfigExtentTitle(@ParentID);
           @Ctx_Title = ConfigContextTitle(ConfigExtentParent(@ParentID));

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
    
