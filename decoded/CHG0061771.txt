
-- CHGXXXXXXX - Change DW4 <9 Customers back to their original Customer ID

-- !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
--    Add manual updates for customers with an "other count" > 0.
-- !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!

-- Build a file containing the <9 Customer ID's for the DW4 Assns and their
-- corresponding/old Cardinal Customer ID.

   Drop table Cathyrhea/dw4cusfix;

   Create   Table cathyrhea/dw4CusFix(
        Crd_Cust    CHAR    (8)    NOT NULL With Default,
        Crd_parent  DECIMAL (2,0)  NOT NULL With Default,
        LIQ_CID     Char    (8)   NOT NULL With Default,
        Xrf_branch  Char    (3)    NOT NULL With Default,
        LIQ_ExtID   Char    (10)   NOT NULL With Default,
        LIQ_Short   Char    (30)   NOT NULL With Default,
        Crd_CIS     Char    (8)    NOT NULL With Default,
        Crd_Short   Char    (20)   NOT NULL With Default,
        Oth_Count   Decimal (5,0)  NOT NULL With Default);

   Label on Table cathyrhea/dw4CusFix is
    'DW4 <9 Customers with old s/b Cardinal CIS';

-- Update the file with DW4 data.

   Insert into cathyrhea/dw4cusfix
   (SELECT CX.CRD_CUST, CX.CRD_PARENT, CX.LIQ_CID,
    substr(CX.LIQ_BRANCH,1,3) as CX_Branch,
    substr(cus_extid,1,8) as XID, substr(CUS_SHTNAM,1,30) as LIQ_short,
    CSMASTR.CSCIS as Card_CIS, CSMASTR.CSSNAM as Card_Short,
    (Select count(*) from OSBLDDATU/bmxcusts cx2 where cx2.liq_cid
     = cx.liq_cid and (cx.crd_parent <> cx2.crd_parent or cx2.crd_cust
     <> cx.crd_cust))
    FROM osblddatu/bmxcusts cx
    join biblddatu/blcusts on liq_cid = cus_cid
    join datubliq/csmastr on cscis = substr(cus_extid,1,8)
    WHERE CRD_PARENT = 05 and
          csdlib = 'DATUS05' and length(rtrim(cus_extid)) between 1
          and 8 and crd_cust like '<9%'
          and exists (Select * from osblddatu/bmxloans lx
          where lx.crd_cust = cx.crd_cust and map_create > '2025-01-01'))
       With NC;

   Insert into cathyrhea/dw4cusfix
   (SELECT CX.CRD_CUST, CX.CRD_PARENT, CX.LIQ_CID,
    substr(CX.LIQ_BRANCH,1,3) as CX_Branch,
    substr(cus_extid,1,8) as XID, substr(CUS_SHTNAM,1,30) as LIQ_short,
    CSMASTR.CSCIS as Card_CIS, CSMASTR.CSSNAM as Card_Short,
    (Select count(*) from OSBLDDATU/bmxcusts cx2 where cx2.liq_cid
     = cx.liq_cid and (cx.crd_parent <> cx2.crd_parent or cx2.crd_cust
     <> cx.crd_cust))
    FROM osblddatu/bmxcusts cx
    join biblddatu/blcusts on liq_cid = cus_cid
    join datubliq/csmastr on cscis = substr(cus_extid,1,8)
    WHERE CRD_PARENT = 08 and
          csdlib = 'DATUS12' and length(rtrim(cus_extid)) between 1
          and 8 and crd_cust like '<9%'
          and exists (Select * from osblddatu/bmxloans lx
          where lx.crd_cust = cx.crd_cust and map_create > '2025-01-01'))
       With NC;

   Insert into cathyrhea/dw4cusfix
   (SELECT CX.CRD_CUST, CX.CRD_PARENT, CX.LIQ_CID,
    substr(CX.LIQ_BRANCH,1,3) as CX_Branch,
    substr(cus_extid,1,8) as XID, substr(CUS_SHTNAM,1,30) as LIQ_short,
    CSMASTR.CSCIS as Card_CIS, CSMASTR.CSSNAM as Card_Short,
    (Select count(*) from OSBLDDATU/bmxcusts cx2 where cx2.liq_cid
     = cx.liq_cid and (cx.crd_parent <> cx2.crd_parent or cx2.crd_cust
     <> cx.crd_cust))
    FROM osblddatu/bmxcusts cx
    join biblddatu/blcusts on liq_cid = cus_cid
    join datubliq/csmastr on cscis = substr(cus_extid,1,8)
    WHERE CRD_PARENT = 18 and
          csdlib = 'DATUS71' and length(rtrim(cus_extid)) between 1
          and 8 and crd_cust like '<9%'
          and exists (Select * from osblddatu/bmxloans lx
          where lx.crd_cust = cx.crd_cust and map_create > '2025-01-01'))
          WITH NC;

-- Use the table to update the BMXCUSTS table for all customers with a zero "other"...
-- replace the <9 with the Cardinal CIS key if it does not already exist.

   update osblddatu/bmxcusts cx
-- update cathyrhea/bmxcusts cx
       Set cx.crd_cust =
       (select dw.crd_cis
       from cathyrhea/dw4cusfix dw where dw.crd_cust = cx.crd_cust
       and dw.oth_count = 0 and dw.crd_parent = cx.crd_parent)
       Where exists
       (select * from cathyrhea/dw4cusfix dw
       where dw.crd_cust = cx.crd_cust
       and dw.oth_count = 0 and dw.crd_parent = cx.crd_parent)
       With NC;

-- Remove the <9 customer from CSMASTR if CSDLIB matches for branch being processed.

   delete from datubliq/csmastr
-- delete from cathyrhea/csmastr
   where cscis like '<9%' and
   not exists
   (select * from osblddatu/bmxcusts where cscis = crd_cust)
-- (select * from cathyrhea/bmxcusts where cscis = crd_cust)
   and exists
   (select * from cathyrhea/dw4cusfix where cscis = crd_cust
      and oth_count = 0)
      With NC;
