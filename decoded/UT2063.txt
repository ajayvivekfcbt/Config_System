     D MaintainOneRow...
     D                 PR                  EXTPGM('UT2028')
     D                                9B 0 const                                Scope ID
     D                                9B 0 const                                Variable Def ID
     D                                2S 0 const                                Option Number
     D                                9B 0 const                                Value ID
     D                                1A                                        End FKey

     D  @IsServer      S              1S 0
     D  @Sort          S            100A
     D  @Type          S              1S 0
     D  @Name          S            100A
     D  @VarDesc       S             50A
     D  @VarVal        S            128A
     D  @Offset        S             10I 0
     D  @LeftLen       S             10I 0
     D  @RightLen      S             10I 0

jse1 D  @SuppressFill  C                   Const(X'1C1C1C1C1C1C1C1C1C')

       Begsr $Load_Subfile;

       // Determine where the data goes

          EXEC SQL
             SELECT   MAX(LENGTH(
                      TRIM(CTX.CTX_NAME) || '.' ||
                      TRIM(XTN.XTN_NAME) || '.' ||
                      TRIM(VDF.VDF_NAME)   ))
             INTO     :@LEFTLEN
             FROM     UTCFGCTX CTX
                         JOIN
                      UTCFGXTN XTN ON XTN.CTX_ID = CTX.CTX_ID
                         JOIN
                      UTCFGVDF VDF ON VDF.XTN_ID = XTN.XTN_ID
                         LEFT JOIN
                      UTCFGVAL VAL ON VAL.VDF_ID = VDF.VDF_ID AND
                                      VAL.SCP_ID = :@Scp_ID
             WHERE    CTX.CTX_IS_SRV = :@IsServer;

             @OffSet = @LeftLen + 3;
             @RightLen = %len(@Text) - @Offset - 42;

       // Retrieve the display data

          EXEC SQL
             DECLARE DISPLAY_CURSOR CURSOR FOR

             SELECT   TRIM(CTX.CTX_NAME) || '.' ||
                      TRIM(XTN.XTN_NAME) || '.' ||
                      TRIM(VDF.VDF_NAME),
                      VDF.VDF_DESC,
                      COALESCE(VAL.VAL_VARVAL,' '),
                      VDF.VDF_ID,
                      COALESCE(VAL.VAL_ID,0)
             FROM     UTCFGCTX CTX
                         JOIN
                      UTCFGXTN XTN ON XTN.CTX_ID = CTX.CTX_ID
                         JOIN
                      UTCFGVDF VDF ON VDF.XTN_ID = XTN.XTN_ID
                         LEFT JOIN
                      UTCFGVAL VAL ON VAL.VDF_ID = VDF.VDF_ID AND
                                      VAL.SCP_ID = :@Scp_ID
             WHERE    CTX.CTX_IS_SRV = :@IsServer
             ORDER BY 1;

       // Open and process the data

          EXEC SQL OPEN DISPLAY_CURSOR;

          DOU SQLCOD <> *zero;

             EXEC SQL FETCH DISPLAY_CURSOR
                INTO  :@NAME, :@VARDESC, :@VARVAL, :@VDF_ID, :@VAL_ID;

             If SQLCOD = *zero;
                Clear @Text;
                %Subst(@Text :  1 : @LeftLen) = %trim(@Name);
                %Subst(@Text :  @OffSet : @LeftLen) = %trim(@VarDesc);
                %Subst(@Text :  86 : 40) = %trim(@VarVal);

jse1            If (%Scan('PASS': Uppercase(@Name)) > *zero or
jse1                %Scan('KEY': Uppercase(@Name)) > *zero ) and
jse1               @VarVal <> *blanks  or
jse1               (%Scan('PASS': Uppercase(@VarDesc)) > *zero or
jse1                %Scan('KEY': Uppercase(@VarDesc)) > *zero ) and
jse1               @VarVal <> *blanks;
jse1                %Subst(@Text : 86 : 40) = *blanks;
jse1                %Subst(@Text : 86 : 12) = @SuppressFill;
jse1            Endif;

                Exsr $Write_Subfile_Record;

             Endif;
          Enddo;

          EXEC SQL CLOSE DISPLAY_CURSOR;

       Endsr;
