     H OPTION(*SRCSTMT : *NODEBUGIO) CVTOPT(*NODATETIME) ALWNULL(*USRCTL)
     H ACTGRP(*CALLER) DFTACTGRP(*NO) BNDDIR('UTBIND')

      //********************************************************************************************
      //
      // Program  : UT2090       UT Config: Maintenance Audit Report Request
      // Project  : Infrastructure - Configuration
      // Author.  : J.Ellis                          Date. . : 09/12/2012
      //
      // Narrative: Provides the means for users to request print of UT Config Maintenance Audit
      //            entries for a date range.
      //
      //********************************************************************************************
      // Modification History:
      //--------------------------------------------------------------------------------------------
      //
      //********************************************************************************************

     FUT2090D   CF   E             WorkStn

     // General Work Fields
     D @NoDate         S               D   Inz(d'0001-01-01') DatFmt(*ISO)
     D @EarliestDate   S               D   Inz(d'0001-01-01') DatFmt(*ISO)
     D @LatestDate     S               D   Inz(d'0001-01-01') DatFmt(*ISO)
     D iError          S               n   Inz(*Off)

     // Procedure Interface Fields
     D #MsgID          S              7A
     D #Command        S            512
     D #Asp            S              1    Inz('''')
     D #MessageText    S            256    Inz(*blanks)                         Message Text or
     D #MessageID      S              7    Inz(*blanks)                         Message ID
     D #MessageData    S             80    Inz(*blanks)                         Message Data Str
     D #MessageFile    S             10    Inz(*blanks)                         Message File

     // Program status data structure
     D                SDS
     D SDS_PgmID               1     10
     D SDS_DeviceID          244    253
     D SDS_UserID            254    263

     // Prototype for Utilities
      /Copy QCPYSRC,UTBIND_PR

      /Free
      //***********************************************************************
       // Main Processing
      //***********************************************************************

      // Process until exit
       Dou *In03;

          // Provide picklist
           Write MSG_CTL;
           Exfmt REQUEST;

          // Clear Subfile Messages
           ClearSubfileMessages(#PgmQ);

          // Process exit request
           If *In03 or (*In12 and not *In90);
               Exsr $Exit;

          // Display messages
           Elseif *In06 and *In95;
               #Command = 'DSPMSG';
               #MsgID = RunCommand(%Trim(#Command) : '*NOLOG');

          // Display workstation submitted jobs
           Elseif *In10 and *In95;
               #Command = 'DSPSBMJOB SBMFROM(*WRKSTN)';
               #MsgID = RunCommand(%Trim(#Command) : '*NOLOG');

          // Cancel submit request and allow date changes
           Elseif *In12 and *In90;
               *In90 = *Off;

          // Submit request
           Elseif *In90;
               *In90 = *Off;
               #Command = 'SBMJOB CMD(CALL PGM(UT2090R) PARM(' + #Asp +
                            %Char(%Date(@DATEFROM:*USA):*ISO)
                            + #Asp + '  ' + #Asp +
                            %Char(%Date(@DATETO:*USA):*ISO)
                            + #Asp +
                            ')) JOB(UT2090R) INLLIBL(*CURRENT)';
               #MsgID = RunCommand(%Trim(#Command) : '*LOG');
               If #MsgID <> *Blanks;
                   #MessageText = 'Submit of report requested FAILED: ' +
                                  %trim(#MsgID) + ' error received.  Contact ' +
                                  'IT Operations for support.';
                   SendSubfileMessage(#PgmQ : %Trim(#MessageText));
               Else;
                   #MessageText = 'Your report request has been submitted.';
                   SendSubfileMessage(#PgmQ : %Trim(#MessageText));
                   *In95 = *On;
               Endif;

          // Process input of date ranges
           Else;

               Exsr $Edit_Input;
               If not iError;
                   *In90 = *On;
                   #MessageText = 'Confirm your date range.  Press ENTER ' +
                                  'to submit request, or F12 to cancel';
                   SendSubfileMessage(#PgmQ : %Trim(#MessageText));
               Endif;

           Endif;
       Enddo;

       // Exit
       Exsr $Exit;

      //***********************************************************************
       // Edit User Input Values
      //***********************************************************************

       Begsr $Edit_Input;

         iError = *Off;
         *In50 = *Off;
         *In51 = *Off;

        //  Edit From Date value
         If @DATEFROM <> *Zero;
             If %Len(%Trim(%Char(@DATEFROM))) <= 6;
                 Test(DE) *MDY @DATEFROM;
             Else;
                 Test(DE) *USA @DATEFROM;
             Endif;
             If %Error;
                 iError = *On;
                 *In50 = *On;
                 #MessageText = 'From Date must be valid ' +
                               'MMDDYYYY or MMDDYY value';
                 SendSubfileMessage(#PgmQ : %Trim(#MessageText));
             Elseif %Len(%Trim(%Char(@DATEFROM))) <= 6;
                 @DATEFROM = %Dec(%Date(@DATEFROM:*MDY):*USA);
             Endif;
         Else;
             iError = *On;
             *In50 = *On;
             #MessageText = 'From Date must be valid MMDDYYYY or MMDDYY value';
             SendSubfileMessage(#PgmQ : %Trim(#MessageText));
         Endif;

        //  Edit To Date value
         If @DATETO <> *Zero;
             If %Len(%Trim(%Char(@DATETO))) <= 6;
                 Test(DE) *MDY @DATETO;
             Else;
                 Test(DE) *USA @DATETO;
             Endif;
             If %Error;
                 iError = *On;
                 *In51 = *On;
                 #MessageText = 'To Date must be valid ' +
                               'MMDDYYYY or MMDDYY value';
                 SendSubfileMessage(#PgmQ : %Trim(#MessageText));
             Elseif %Len(%Trim(%Char(@DATETO))) <= 6;
                 @DATETO = %Dec(%Date(@DATETO:*MDY):*USA);
             Endif;
         Else;
             iError = *On;
             *In51 = *On;
             #MessageText = 'To Date must be valid MMDDYYYY or MMDDYY value';
             SendSubfileMessage(#PgmQ : %Trim(#MessageText));
         Endif;

        //  From date can not be greater than to date
         If not iError and %Date(@DATEFROM:*USA) > %Date(@DATETO:*USA);
             iError = *On;
             *In50 = *On;
             #MessageText = 'From Date can not be after To Date';
             SendSubfileMessage(#PgmQ : %Trim(#MessageText));
         Endif;

        //  Get out if basic value error has occurred
         If iError;
             LeaveSr;
         Endif;


       Endsr;

      //***********************************************************************
       //Program Exit
      //*************************************************************************

       Begsr $Exit;

         *InLR = *On;
         Return;

       Endsr;

      //***********************************************************************
       //Program Initialization
      //***********************************************************************

       Begsr *InzSr;
    
        // Get earliest audit entry date available
         Exec SQL
              Select  Date(AUD_LOGGED)
              Into    :@EarliestDate
              From    UTCFGAUD
              Order by AUD_LOGGED
              Fetch First Row Only;
    
        // Get latest audit entry date available
         Exec SQL
              Select  Date(AUD_LOGGED)
              Into    :@LatestDate
              From    UTCFGAUD
              Order by AUD_LOGGED Desc
              Fetch First Row Only;
    
        // Setup messaging subfile and clear
         #PgmQ = SDS_PgmID;
         ClearSubfileMessages(#PgmQ);
    
        // Set first time display date range for what is available
         @DATEFROM = %Dec(@EarliestDate:*USA);
         @DATETO = %Dec(@LatestDate:*USA);
         #MessageText = 'The dates displayed is the date range available ' +
                        'for reporting';
         SendSubfileMessage(#PgmQ : %Trim(#MessageText));

       Endsr;

      /End-Free
