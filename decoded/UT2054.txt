       Begsr $Load_Subfile;

       // Retrieve the data for display

           EXEC SQL
               DECLARE LIST_CURSOR CURSOR FOR
               SELECT   CTX.CTX_ID, CTX.CTX_NAME, CTX.CTX_DESC,
                        BIT_YESNO(CTX.CTX_IS_SRV)
               FROM     UTCFGSCP SCP
                            JOIN
                        UTCFGCTX CTX ON CTX.CTX_IS_SRV = SCP.SCP_IS_SRV
               WHERE    SCP.SCP_ID = :@ScopeID
               ORDER BY UPPER(CTX.CTX_NAME);

           EXEC SQL OPEN LIST_CURSOR;

           DOU SQLCOD <> *zero;

               EXEC SQL
                   FETCH LIST_CURSOR
                   INTO  :@ID, :@NAME, :@DESC, :@SRV_YN;

               If SQLCOD = *zero;
                   Exsr $Write_Subfile_Record;
               Endif;
           Enddo;

           EXEC SQL CLOSE LIST_CURSOR;
