       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Ctx_Name   = TrimAndCheck(%trimr(Ctx_Name  ) : *IN95);
           Ctx_Desc   = TrimAndCheck(%trimr(Ctx_Desc  ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Ctx_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must meet configuration standards

           If @ErrorMsg = *blank and
               ConfigValidName(Ctx_Name) <> *ON;
               @ErrorMsg = 'Name does not meet configuration standards';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigContextID(Ctx_Name) <> *zero and
               ConfigContextID(Ctx_Name) <> @Id) or
               (@Option <> 2 and
               ConfigContextID(Ctx_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Ctx_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       // Server level must be YES or NO

           If @ErrorMsg = *blank and
               @YN_Is_Srv <> 'YES' and
               @YN_Is_Srv <> 'NO';
               @ErrorMsg = 'Server Level must be YES or NO';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Load screen values to database values

           Ctx_Info = %trim(@Info);
           Ctx_Is_Srv = YesNoToBit(@YN_Is_Srv);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigContextInsert(@DB);
           ElseIf @Option = 2;
               ConfigContextUpdate(@DB);
           ElseIf @Option = 4;
               ConfigContextDelete(@DB);
           Endif;

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Prompt;

       // check for prompt on field where cursor is positioned

           @ErrorMsg = 'Prompt function not supported or defined ' +
                       'at current cursor position';

       Endsr;
    
       //---------------------------------------------------------------------

       Begsr $Check_Function_Key_Requests;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
               @DB = ConfigContextGet(@Id);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGCTX';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';

       // Check for delete with child records

           If @Option = 4 and
              ConfigContextKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info      = Ctx_Info;
           @YN_Is_Srv = BitToYesNo(Ctx_Is_Srv);

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
