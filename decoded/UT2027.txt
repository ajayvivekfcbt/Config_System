     D                                9B 0 const                                Parent ID
     D                                2S 0 const                                Option Number
     D                                9B 0 const                                Row ID
     D                                1A                                        End FKey

     D EntryParms      PI
     D   @ParentID                    9B 0 const                                Parent ID
     D   @Option                      2S 0 const                                Option Number
     D   @ID                          9B 0 const                                Row ID
     D   @EndFKey                     1A                                        End FKey

       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           @Value     = TrimAndCheck(%trimr(@Value    ) : *IN95);
           Vvl_Desc   = TrimAndCheck(%trimr(Vvl_Desc  ) : *IN95);

       // Value is required

           If @ErrorMsg = *blank and
               @Value = *blank;
               @ErrorMsg = 'Value is a required entry';
           Endif;

       // Value must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigVarValidID(@ParentID : @Value) <> *zero and
               ConfigVarValidID(@ParentID : @Value) <> @Id) or
               (@Option <> 2 and
               ConfigVarValidID(@ParentID : @Value) <> *zero);
               @ErrorMsg = 'Value is a duplicate entry';
           Endif;

       // Description is Required

           If @ErrorMsg = *blank and
               Vvl_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Prime parent key value

           Vdf_ID = @ParentID;

       // Load screen values to database values

           Vvl_Info = %trim(@Info);
           Vvl_Value =%trim(@Value);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigVarValidInsert(@DB);
           ElseIf @Option = 2;
               ConfigVarValidUpdate(@DB);
           ElseIf @Option = 4;
               ConfigVarValidDelete(@DB);
           Endif;

       Endsr;
    

                       'at current cursor position';

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
               @DB = ConfigVarValidGet(@Id);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGVVL';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';

       // Translate stored into displayed values

           @Info      = Vvl_Info;
           @Value     = Vvl_Value;

       // Set header values

           @Vdf_Title = ConfigVarDefTitle(@ParentID);
           @Xtn_Title = ConfigExtentTitle(
                            ConfigVarDefParent(@ParentID));
           @Ctx_Title = ConfigContextTitle(
                            ConfigExtentParent(
                                ConfigVarDefParent(@ParentID)));

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
