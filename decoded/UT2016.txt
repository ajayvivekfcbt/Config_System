       Begsr $Load_Subfile;

       // Retrieve the data for display

           EXEC SQL
               DECLARE LIST_CURSOR CURSOR FOR
               SELECT   VDF_ID, VDF_NAME, VDF_DESC,
                        BIT_YESNO(VDF_RSTRCT)
               FROM     UTCFGVDF
               WHERE    XTN_ID = :@ParentID
               ORDER BY UPPER(VDF_NAME);

           EXEC SQL OPEN LIST_CURSOR;

           DOU SQLCOD <> *zero;

               EXEC SQL
                   FETCH LIST_CURSOR
                   INTO  :@ID, :@NAME, :@DESC, :@RSTR_YN;

               If SQLCOD = *zero;
                   Exsr $Write_Subfile_Record;
               Endif;
           Enddo;

           EXEC SQL CLOSE LIST_CURSOR;

       Endsr;

           If @Option = 8;
               InfoWindow(ConfigVarDefTitle(@ID) : ConfigVarDefInfo(@ID));
           Elseif @Option = 7;
               If @Rstr_YN = 'YES';
                  WorkWithChildren(@ID : @EndFKey);
                  #WS.FunctionKeyPressed = @EndFKey;
               Else;
               @ErrorMsg = 'Option 7 is only allowed if variable is ' +
                           'defined with restricted values.';
               Endif;
           Elseif @Option = 2 or
               @Option = 3 or
               @Option = 4 or
               @Option = 5;
               MaintainOneRow(@ParentID : @Option : @ID : @EndFKey);
               #WS.FunctionKeyPressed = @EndFKey;
           Else;
               @ErrorMsg = 'Option ' + %Char(@Option) +
                           ' is defined but not supported by code';
