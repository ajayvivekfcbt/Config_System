           EXEC SQL
               DECLARE LIST_CURSOR CURSOR FOR
               SELECT   VDF.VDF_ID,
                        COALESCE(VAL.VAL_ID,0),
                        VDF.VDF_NAME,
                        COALESCE(VAL.VAL_VARVAL,' ')
               FROM     UTCFGVDF VDF
                            LEFT JOIN
                        UTCFGVAL VAL ON VAL.VDF_ID = VDF.VDF_ID AND
                                        VAL.SCP_ID = :@ScopeID
               WHERE    VDF.XTN_ID = :@ParentID
               ORDER BY UPPER(VDF.VDF_NAME);

           EXEC SQL OPEN LIST_CURSOR;

           DOU SQLCOD <> *zero;

               EXEC SQL
                   FETCH LIST_CURSOR
                   INTO  :@VARDEFID, :@VALUEID, :@NAME, :@VALUE;

jse1               If (%Scan('PASS': Uppercase(@Name)) > *zero or
