       Begsr $One_Row_Edit;

       // Reset error conditions

           Clear @ErrorMsg;

       // Left justify all alpha fields

           Vtp_Name   = TrimAndCheck(%trimr(Vtp_Name  ) : *IN95);
           Vtp_Desc   = TrimAndCheck(%trimr(Vtp_Desc  ) : *IN95);

       // Name is required

           If @ErrorMsg = *blank and
               Vtp_Name = *blank;
               @ErrorMsg = 'Name is a required entry';
           Endif;

       // Name must be unique

           If @ErrorMsg = *blank and
               (@Option = 2 and
               ConfigVarTypeID(Vtp_Name) <> *zero and
               ConfigVarTypeID(Vtp_Name) <> @Id) or
               (@Option <> 2 and
               ConfigVarTypeID(Vtp_Name) <> *zero);
               @ErrorMsg = 'Name is a duplicate entry';
           Endif;

       // Description is required

           If @ErrorMsg = *blank and
               Vtp_Desc = *blank;
               @ErrorMsg = 'Description is a required entry';
           Endif;

       Endsr;
    
       //--------------------------------------------------------------------

       Begsr $One_Row_Update;

       // Load screen values to database values

           Vtp_Proc = %trimr(@Proc);
           Vtp_Info = %trimr(@Info);

       // Execute method appropriate for action

           If @Option = *zero or
               @Option = 3;
               ConfigVarTypeInsert(@DB);
           ElseIf @Option = 2;
               ConfigVarTypeUpdate(@DB);
           ElseIf @Option = 4;
               ConfigVarTypeDelete(@DB);
           Endif;

       Endsr;
    

          @ErrorMsg = 'Prompt function not supported or defined ' +
                      'at current cursor position';

       Begsr $Data_Initialization;

       // If @Id received - get row

           If @Id <> *Zero;
               @DB = ConfigVarTypeGet(@Id);
           Endif;

       // Set tables and rights required

           @TableNames(1)  = 'UTCFGVTP';
           @TableRights(1) = 'CRUD';

       // Set valid function keys

           @AvlFnc(3)  = 'F3=Exit';
           @AvlFnc(5)  = 'F5=Refresh';
           @AvlFnc(12) = 'F12=Cancel';

       // Check for delete with child records

           If @Option = 4 and
              ConfigVarTypeKids(@Id) > *zero;
               Write OneRow_D;
               DeleteRestricted();
               #WS.FunctionKeyPressed = #Cancel;
               Exsr $Abort;
           Endif;

       // Translate stored into displayed values

           @Info = Vtp_Info;
           @Proc = Vtp_Proc;

       // Set refresh flag off

           @Refresh = *off;

       Endsr;
