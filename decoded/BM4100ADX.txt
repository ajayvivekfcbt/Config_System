         i = 1;
         dou i < 22 ;
           sqlstmt = 'merge into bmbmdatad '  +
                     'using  bmxloans '      +
                     ' on    crd_loan = BM_ACCTNUM '    +
                     ' and MAP_MASTER = ''Y'' '      +
                     ' and bm_cricod  = ''' + %trim(criteria(i)) + '''' +
                     ' and MAP_DROP = date(''01/01/0001'') ' +
                     ' when matched then update '       +
                     'set BM_DEALID = lIQ_deal,BM_FACLTY = liq_faclty, '  +
                     'BM_OSTRID = LIQ_OSTRID  , BM_EXPCDE = LIQ_EXPCDE ' +
                     ', BM_PORTFO = LIQ_PORTFO ' ;

           exec sql execute immediate :sqlstmt;

           sqlstmt = ' ';
           sqlstmt = 'merge into bmbmdatad '  +
                     ' using  bmxdepos '      +
                     ' on   CRD_ACCTNO = BM_ACCTNUM '    +
                     ' and bm_cricod  = ''' + %trim(criteria(i)) + '''' +
                     ' when matched then update '       +
                     ' set BM_DEALID = LIQ_DEALID,BM_FACLTY = LIQ_FACRID, '  +
                     ' BM_OSTRID = LIQ_OSTRID  , BM_EXPCDE = LIQ_EXPCDE ' +
                     ', BM_PORTFO = LIQ_PORTFO ' ;

           exec sql execute immediate :sqlstmt;

           sqlstmt = ' ';
           sqlstmt = 'merge into bmbmdatad '  +
                     ' using  BMXSTOCK '      +
                     ' on   CRD_ACCTNO = BM_ACCTNUM '    +
                     ' and bm_cricod  = ''' + %trim(criteria(i)) + '''' +
                     ' when matched then update '       +
                     ' set BM_DEALID = LIQ_DEALID,BM_FACLTY = LIQ_FACRID, '  +
                     ' BM_OSTRID = LIQ_OSTRID  , BM_EXPCDE = LIQ_EXPCDE ' ;

           exec sql execute immediate :sqlstmt;

           i+=1;
          enddo;
          *inlr = *on;
